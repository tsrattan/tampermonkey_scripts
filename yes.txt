javascript:(function () {
  const wait = ms => new Promise(r => setTimeout(r, ms));

  function setInputValueIfExists(selector, value) {
    const el = document.querySelector(selector);
    if (!el) return false;

    try {
      el.focus();

      const proto =
        el.tagName === 'TEXTAREA'
          ? HTMLTextAreaElement.prototype
          : HTMLInputElement.prototype;

      const setter = Object.getOwnPropertyDescriptor(proto, 'value')?.set;
      if (setter) setter.call(el, value);
      else el.value = value;

      el.dispatchEvent(new Event('input', { bubbles: true }));
      el.dispatchEvent(new Event('change', { bubbles: true }));
      el.blur();
      return true;
    } catch (e) {
      return false;
    }
  }

  async function pickDropdownOptionIfExists(text, dropdownIndex = 0) {
    try {
      const dropdowns = [...document.querySelectorAll('[aria-haspopup="listbox"]')];
      const dropdown = dropdowns[dropdownIndex] || dropdowns[0];
      if (!dropdown) return false;

      dropdown.dispatchEvent(new MouseEvent('mousedown', { bubbles: true }));
      dropdown.click();
      await wait(700);

      const option = [...document.querySelectorAll('[role="option"], li, div, span')]
        .find(el => el.textContent && el.textContent.trim() === text);

      if (!option) {
        document.body.click();
        return false;
      }

      option.dispatchEvent(new MouseEvent('mousedown', { bubbles: true }));
      option.click();
      await wait(700);
      return true;
    } catch (e) {
      return false;
    }
  }

  function clickYesButtonsIfExists() {
    const buttons = document.querySelectorAll('[data-automation-value="Yes"]');
    if (!buttons.length) return false;
    buttons.forEach(el => el.click());
    return true;
  }

  async function runAvailableItems() {
    setInputValueIfExists('input[data-automation-id="textInput"]', 'Rattan Tajinder');
    await wait(500);

    await pickDropdownOptionIfExists('APAC', 0);
    await wait(500);

    clickYesButtonsIfExists();
  }

  let syd27Done = false;
  let attempts = 0;
  const maxAttempts = 120;

  async function watchForLaterItems() {
    if (syd27Done) return;

    attempts++;
    if (attempts > maxAttempts) {
      clearInterval(window.__formWatcher);
      return;
    }

    const success = await pickDropdownOptionIfExists('SYD27', 0);
    if (success) {
      await wait(500);
      clickYesButtonsIfExists();
      syd27Done = true;
      clearInterval(window.__formWatcher);
      alert('Available items filled. SYD27 selected if present.');
    }
  }

  runAvailableItems();

  if (window.__formWatcher) clearInterval(window.__formWatcher);
  window.__formWatcher = setInterval(watchForLaterItems, 1500);

  alert('Running: it will fill only fields that exist on this form.');
})();
