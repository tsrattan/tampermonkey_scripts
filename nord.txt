javascript:(()=>{
  const id='nord-planning-calendar';
  const attr='data-nord-planning';
  const previous=document.getElementById(id);
  if(previous){
    previous.remove();
    document.querySelectorAll('['+attr+']').forEach(el=>el.removeAttribute(attr));
    return;
  }
  const rgb=value=>{
    const match=value.match(/^rgba?\(([^)]+)\)$/);
    return match ? match[1].split(',').map(Number) : null;
  };
  const neutral=c=>c && Math.max(...c.slice(0,3))-Math.min(...c.slice(0,3))<=14;
  const surfaces=new WeakMap();
  const groups=new Map();
  const rules=[];
  for(const el of document.querySelectorAll('html,body,body *')){
    if(!(el instanceof HTMLElement)) continue;
    const css=getComputedStyle(el);
    const bg=rgb(css.backgroundColor);
    const fg=rgb(css.color);
    const solid=bg && (bg.length===3 || bg[3]>=0.95);
    const painted=css.backgroundImage!=='none';
    const lightNeutral=solid && neutral(bg) && bg[0]>=155;
    const ownSurface=solid || painted;
    const nord=ownSurface ? lightNeutral && !painted : (surfaces.get(el.parentElement) ?? true);
    surfaces.set(el,nord);
    const changes=[];
    if(nord && (lightNeutral || el===document.documentElement)){
      const tone=lightNeutral ? bg[0] : 255;
      const shade=tone>=252?'#2E3440':tone>=239?'#353C4A':tone>=218?'#414A5A':'#4C566A';
      changes.push('background-color:'+shade+'!important');
    }
    if(nord && fg){
      if(el.closest('a[href]')) changes.push('color:#88C0D0!important');
      else if(neutral(fg)) changes.push('color:'+(fg[0]<100?'#ECEFF4':'#D8DEE9')+'!important');
    }else if(!nord && fg){
      changes.push('color:'+css.color+'!important');
    }
    if(nord){
      for(const side of ['top','right','bottom','left']){
        const key='border-'+side;
        const border=rgb(css.getPropertyValue(key+'-color'));
        if(css.getPropertyValue(key+'-width')!=='0px' && neutral(border) && border[0]>=150){
          changes.push(key+'-color:#4C566A!important');
        }
      }
    }
    if(!changes.length) continue;
    const declaration=changes.join(';');
    if(!groups.has(declaration)){
      const group=String(groups.size);
      groups.set(declaration,group);
      rules.push('['+attr+'="'+group+'"]{'+declaration+'}');
    }
    el.setAttribute(attr,groups.get(declaration));
  }
  const style=document.createElement('style');
  style.id=id;
  style.textContent=rules.join('\n');
  document.head.append(style);
})()
